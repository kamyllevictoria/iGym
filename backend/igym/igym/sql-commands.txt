
CREATE TABLE usuario (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    email VARCHAR(200) NOT NULL UNIQUE,
    senha VARCHAR(10) NOT NULL,
    telefone VARCHAR(11) NOT NULL,
    sexo VARCHAR(10) NOT NULL,
    dataNascimento DATE NOT NULL,
    cpf VARCHAR(15) NOT NULL UNIQUE,
    idade INT NOT NULL
);


CREATE TABLE aluno (
    matricula BIGINT PRIMARY KEY DEFAULT floor(random() * 10000000000)::bigint,
    usuario_id INT NOT NULL UNIQUE,
    altura NUMERIC(5,2) NOT NULL,
    peso NUMERIC(5,2) NOT NULL,
    medicamentos VARCHAR(200),
    cirurgias VARCHAR(200),
    metodo_pagamento VARCHAR(50) NOT NULL CHECK (metodo_pagamento IN ('CREDITO','DEBITO','PIX','DINHEIRO')),
    pressao_arterial VARCHAR(3) NOT NULL,
    historico_de_saude VARCHAR(255) NOT NULL,
    ativo BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (usuario_id) REFERENCES usuario(id) ON DELETE CASCADE

);

CREATE TABLE professor (
    cref VARCHAR(7) PRIMARY KEY,
    usuario_id INT NOT NULL UNIQUE,
    FOREIGN KEY (usuario_id) REFERENCES usuario(id) ON DELETE CASCADE
);


create table fichaDeTreino(
	id serial primary key,
	crefProfessor varchar(7) not null,
	matriculaAluno bigint not null,
	dataCriacao date not null,
	frequenciaSemanal int not null,
	foreign key (crefProfessor) references professor(cref)
		on delete cascade
		on update cascade,
	foreign key (matriculaAluno) references aluno(matricula)
		on delete cascade
		on update cascade
);

create table exercicios(
	id serial primary key,
	fichaDeTreinoId int not null,
	nome varchar(100) not null,
	agrupamentoMuscular varchar(50),
	series int not null,
	repeticoes int not null,
	tempoDeDescanso numeric(3,2),
	foreign key (fichaDeTreinoId) references fichaDeTreino(id)
		on delete cascade
		on update cascade
);


SELECT table_name
FROM information_schema.tables
WHERE table_schema = 'public'
AND table_type = 'BASE TABLE'
ORDER BY table_name;


--join professor e usuario

    *

INNER JOIN usuario U
ON P.usuarioid = U.id;
