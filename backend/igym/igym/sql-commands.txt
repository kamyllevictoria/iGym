CREATE TABLE usuario (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(255) NOT NULL,
    email VARCHAR(200) NOT NULL,
    senha VARCHAR(60) NOT NULL UNIQUE,
    telefone VARCHAR(11) NOT NULL,
    genero VARCHAR(10) NOT NULL CHECK (genero IN ('FEMININO','MASCULINO','OUTROS')),
    dataNascimento DATE NOT NULL,
    cpf VARCHAR(15) NOT NULL UNIQUE,
    idade INT NOT NULL
);


CREATE TABLE aluno (
    matricula SERIAL PRIMARY KEY,
    altura INT NOT NULL,
    peso NUMERIC(5,2) NOT NULL,
    medicamentos VARCHAR(200),
    cirurgias VARCHAR(200),
    metodoPagamento VARCHAR(50) NOT NULL CHECK (metodoPagamento IN ('CREDITO','DEBITO','PIX','DINHEIRO')),
    pressaoArterial VARCHAR(3) NOT NULL,
    historicoSaude VARCHAR(255) NOT NULL,
    plano CHECK (metodoPagamento IN ('ATIVO','INATIVO')),
    FOREIGN KEY (usuarioId) REFERENCES usuario(id) ON DELETE CASCADE
);

CREATE TABLE professor (
    cref VARCHAR(7) PRIMARY KEY,
    FOREIGN KEY (usuarioId) REFERENCES usuario(id) ON DELETE CASCADE
);


create table fichaDeTreino(
	id SERIAL PRIMARY KEY,
	dataCriacao DATE NOT NULL,
	frequenciaSemanal INT NOT NULL,
	foreign key (crefProfessor) references professor(cref)
		on delete cascade
		on update cascade,
	foreign key (matriculaAluno) references aluno(matricula)
		on delete cascade
		on update cascade
);

create table exercicios(
	id SERIAL PRIMARY KEY,
	nome varchar(100) NOT NULL,
	agrupamentoMuscular varchar(50),
	series INT NOT NULL,
	repeticoes INT NOT NULL,
	tempoDeDescanso NUMERIC NOT NULL(3,2),
	foreign key (fichaDeTreinoId) references fichaDeTreino(id)
		on delete cascade
		on update cascade
);

create table administrador(
    email varchar(200) NOT NULL,
    senha varchar(60) NOT NULL
)


--join professor e usuario
SELECT
    *
FROM professor P
INNER JOIN usuario U
ON P.usuarioid = U.id;

SELECT
    *
FROM aluno A
INNER JOIN usuario U
ON A.usuarioid = U.id;
